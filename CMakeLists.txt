cmake_minimum_required(VERSION 3.16)
project(focusguard VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# Windows: icon and version information for an executable (src/focusguard.rc.in).
function(focusguard_resources target name description)
  if(WIN32)
    set(FG_ICON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/data")
    set(FG_RC_NAME "${name}")
    set(FG_RC_DESCRIPTION "${description}")
    configure_file(src/focusguard.rc.in "${CMAKE_CURRENT_BINARY_DIR}/${target}.rc" @ONLY)
    target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}.rc")
  endif()
endfunction()

if(WIN32)
  option(FOCUSGUARD_NO_CONSOLE "Build as a GUI-subsystem app (tray only, no console window)" OFF)
  add_executable(focusguard src/focusguard_win.cpp src/common.hpp)
  target_link_libraries(focusguard PRIVATE user32 shell32 dwmapi advapi32)
  focusguard_resources(focusguard focusguard "FocusGuard - stops windows from stealing keyboard focus")
  if(FOCUSGUARD_NO_CONSOLE)
    set_target_properties(focusguard PROPERTIES WIN32_EXECUTABLE ON)
    if(MSVC)
      target_link_options(focusguard PRIVATE /ENTRY:mainCRTStartup)
    endif()
  endif()
  if(MSVC)
    target_compile_options(focusguard PRIVATE /W4 /utf-8)
    set_property(TARGET focusguard PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(focusguard PRIVATE -Wall -Wextra)
    target_link_options(focusguard PRIVATE -static)
  endif()
else()
  find_package(X11 REQUIRED)
  if(NOT X11_Xi_FOUND)
    message(FATAL_ERROR "libXi headers not found: install libxi-dev (Debian/Ubuntu), libXi-devel (Fedora) or libxi (Arch)")
  endif()
  add_executable(focusguard src/focusguard_x11.cpp src/common.hpp src/tray_sni.hpp)
  target_link_libraries(focusguard PRIVATE X11::X11 X11::Xi)

  option(FOCUSGUARD_TRAY "Tray icon and D-Bus notifications (needs libdbus-1)" ON)
  if(FOCUSGUARD_TRAY)
    find_package(PkgConfig)
    if(PKG_CONFIG_FOUND)
      pkg_check_modules(DBUS IMPORTED_TARGET dbus-1)
    endif()
    if(NOT DBUS_FOUND)
      message(FATAL_ERROR "libdbus-1 not found: install libdbus-1-dev (Debian/Ubuntu), dbus-devel (Fedora) "
                          "or dbus (Arch), or configure with -DFOCUSGUARD_TRAY=OFF to build without the tray icon")
    endif()
    target_link_libraries(focusguard PRIVATE PkgConfig::DBUS)
    target_compile_definitions(focusguard PRIVATE FOCUSGUARD_TRAY)
  endif()
  target_compile_options(focusguard PRIVATE -Wall -Wextra)
  install(TARGETS focusguard RUNTIME DESTINATION bin)
endif()

# ---------------------------------------------------------------- tests

option(FOCUSGUARD_TESTS "Build the unit tests (run them with ctest)" ON)
if(FOCUSGUARD_TESTS)
  enable_testing()
  add_executable(focusguard-tests tests/tests.cpp src/common.hpp src/control.hpp)
  target_include_directories(focusguard-tests PRIVATE src)
  if(MSVC)
    target_compile_options(focusguard-tests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(focusguard-tests PRIVATE -Wall -Wextra)
  endif()
  add_test(NAME unit-tests COMMAND focusguard-tests)
endif()

# Single source of truth for the version (see FOCUSGUARD_VERSION in common.hpp).
target_compile_definitions(focusguard PRIVATE FOCUSGUARD_VERSION="${PROJECT_VERSION}")

# ---------------------------------------------------------------- settings app

option(FOCUSGUARD_GUI "Build the FocusGuard Settings app (needs Qt 6.2+ Widgets and Network)" ON)
if(FOCUSGUARD_GUI)
  find_package(Qt6 6.2 QUIET COMPONENTS Widgets Network)
  if(NOT Qt6_FOUND)
    message(WARNING "Qt 6 not found: skipping the settings app. Install qt6-base-dev (Debian/Ubuntu), "
                    "qt6-qtbase-devel (Fedora) or qt6-base (Arch), point CMAKE_PREFIX_PATH at Qt, "
                    "or configure with -DFOCUSGUARD_GUI=OFF.")
  else()
    add_executable(focusguard-settings WIN32
      gui/main.cpp gui/main_window.cpp gui/main_window.h gui/pages.cpp gui/pages.h
      gui/widgets.cpp gui/widgets.h gui/daemon_client.cpp gui/daemon_client.h
      gui/platform.cpp gui/platform.h gui/settings_model.cpp gui/settings_model.h
      src/common.hpp src/control.hpp)
    set_target_properties(focusguard-settings PROPERTIES AUTOMOC ON)
    target_include_directories(focusguard-settings PRIVATE src)
    target_link_libraries(focusguard-settings PRIVATE Qt6::Widgets Qt6::Network)
    focusguard_resources(focusguard-settings focusguard-settings "FocusGuard Settings")
    target_compile_definitions(focusguard-settings PRIVATE FOCUSGUARD_VERSION="${PROJECT_VERSION}")
    if(MSVC)
      target_compile_options(focusguard-settings PRIVATE /W4 /utf-8)
    else()
      target_compile_options(focusguard-settings PRIVATE -Wall -Wextra)
    endif()
    if(NOT WIN32)
      install(TARGETS focusguard-settings RUNTIME DESTINATION bin)
      install(FILES data/focusguard-settings.desktop DESTINATION share/applications)
      install(FILES data/focusguard.svg DESTINATION share/icons/hicolor/scalable/apps)
    endif()
  endif()
endif()
